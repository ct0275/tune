-- db별 wraparound
SELECT
    datname,
    age(datfrozenxid) AS xid_age,
    ROUND(
        age(datfrozenxid)::numeric /
        current_setting('autovacuum_freeze_max_age')::numeric * 100,
        2
    ) AS usage_pct
FROM pg_database
ORDER BY age(datfrozenxid) DESC;

-- table별 wraparound
SELECT
     n.nspname AS schema_name,
     c.relname AS table_name,
     age(c.relfrozenxid) AS xid_age,
     pg_size_pretty(pg_total_relation_size(c.oid)) AS total_size,
     c.reltuples::bigint AS estimated_rows
 FROM pg_class c
 JOIN pg_namespace n
     ON n.oid = c.relnamespace
 WHERE c.relkind IN ('r','m')
 ORDER BY age(c.relfrozenxid) DESC
 LIMIT 3;

SELECT
     max(age(c.relfrozenxid)) AS max_xid_age,
     avg(age(c.relfrozenxid))::bigint AS avg_xid_age,
     count(*) FILTER (WHERE age(c.relfrozenxid) >= 190000000) AS over_190m,
     count(*) FILTER (WHERE age(c.relfrozenxid) >= 195000000) AS over_195m,
     count(*) FILTER (WHERE age(c.relfrozenxid) >= 199000000) AS over_199m
 FROM pg_class c
 WHERE c.relkind IN ('r','m');
